# FastSCP 实时进度 Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Add real-time transfer progress (panel + quick-send HUD) to FastSCP, computed as transferred bytes / total bytes, with a single shared `TransferTracker` driving both UI surfaces.

**Architecture:** A pure state machine `TransferAggregator` (in `FastSCPCore`) consumes parsed scp progress lines (percent + filename + rate) and maintains a monotonic byte-based total. A `@MainActor` `TransferTracker` wraps it with a background `FileManager` pre-scan and publishes to SwiftUI. The popup (`DestinationPanel`) swaps to a transfer-status view; the quick-send path creates a custom top-right `NSPanel` HUD that auto-closes on success.

**Tech Stack:** Swift 6, SwiftUI, AppKit (`NSPanel`, `NSVisualEffectView`), `FileManager.enumerator`, `Process` (existing `SSHExecutor`), xcodegen (`project.yml`), XCTest.

**Design doc:** `docs/plans/2026-08-07-real-time-progress-design.md`

```swift
import Foundation

public struct SCPProgress: Equatable, Sendable {
    public let percent: Int
    public let fileName: String?
    public let rateBytesPerSec: Int64?
    public let detail: String

    public init(percent: Int, fileName: String?, rateBytesPerSec: Int64?, detail: String) {
        self.percent = percent
        self.fileName = fileName
        self.rateBytesPerSec = rateBytesPerSec
        self.detail = detail
    }
}

public enum SCPProgressParser {
    /// Best-effort parse of an OpenSSH `scp` stderr progress line:
    /// "\r<name><spaces>NNN%<size> <rate> <eta>".
    /// Returns nil if the line carries no percent at all.
    public static func parse(_ line: String) -> SCPProgress? {
        // Strip a leading \r so filenames with leading whitespace don't confuse trim.
        let stripped = line.hasPrefix("\r") ? String(line.dropFirst()) : line

        guard let pctRange = stripped.range(of: #"\b\d{1,3}%"#, options: .regularExpression) else {
            return nil
        }
        let numberText = String(stripped[pctRange].dropLast())
        guard let pct = Int(numberText), (0...100).contains(pct) else { return nil }

        let fileName = String(stripped[..<pctRange.lowerBound])
            .trimmingCharacters(in: .whitespacesAndNewlines)

        let detail = String(stripped[pctRange.upperBound...])
            .trimmingCharacters(in: .whitespacesAndNewlines)

        let rateBytesPerSec = parseRate(from: detail)

        return SCPProgress(
            percent: pct,
            fileName: fileName.isEmpty ? nil : fileName,
            rateBytesPerSec: rateBytesPerSec,
            detail: detail
        )
    }

    /// Parse "12.3MB/s", "456KB/s", "1.2GB/s" into bytes/second. Returns nil if absent.
    static func parseRate(from detail: String) -> Int64? {
        guard let r = detail.range(of: #"(\d+(?:\.\d+)?)\s*(K|M|G)?B/s"#, options: .regularExpression)
        else { return nil }
        let token = detail[r]
        // Scan leading digits/dot for the numeric value; capture optional K/M/G unit.
        var chars: [Character] = []
        var unit: Character = "B"
        for c in token {
            if c.isNumber || c == "." {
                chars.append(c)
            } else if c == "K" || c == "M" || c == "G" {
                unit = c
                break
            } else if !chars.isEmpty {
                break
            }
        }
        guard let n = Double(String(chars)), n >= 0 else { return nil }
        let mult: Double
        switch unit {
        case "K": mult = 1024
        case "M": mult = 1024 * 1024
        case "G": mult = 1024 * 1024 * 1024
        default:  mult = 1
        }
        return Int64(n * mult)
    }
}
```

> Note: keep this parser self-contained — `parseRate` is `internal` (default), called only by `parse`.

```swift
import Foundation

public enum ByteFormat {
    public static func size(_ bytes: Int64) -> String {
        let b = max(bytes, 0)
        if b < 1024 { return "\(b) B" }
        let kb = Double(b) / 1024
        if kb < 1024 { return String(format: "%.1f KB", kb) }
        let mb = kb / 1024
        if mb < 1024 { return String(format: "%.1f MB", mb) }
        let gb = mb / 1024
        return String(format: "%.2f GB", gb)
    }

    public static func rate(_ bytesPerSec: Int64) -> String {
        size(bytesPerSec) + "/s"
    }
}
```

```swift
import Foundation

public enum TransferPhase: Equatable, Sendable {
    case preparing
    case sending
    case done
    case failed(String)
}

public struct TransferProgress: Equatable, Sendable {
    public let phase: TransferPhase
    public let totalBytes: Int64
    public let completedBytes: Int64
    public let totalFiles: Int
    public let currentFileIndex: Int
    public let currentFileName: String?
    public let rateBytesPerSec: Int64?
    public let etaSeconds: Int?

    public var percent: Double {
        guard totalBytes > 0 else { return phase == .done ? 1 : 0 }
        let raw = Double(completedBytes) / Double(totalBytes)
        return min(max(raw, 0), 1)
    }

    public init(phase: TransferPhase,
                totalBytes: Int64,
                completedBytes: Int64,
                totalFiles: Int,
                currentFileIndex: Int,
                currentFileName: String?,
                rateBytesPerSec: Int64?,
                etaSeconds: Int?) {
        self.phase = phase
        self.totalBytes = totalBytes
        self.completedBytes = completedBytes
        self.totalFiles = totalFiles
        self.currentFileIndex = currentFileIndex
        self.currentFileName = currentFileName
        self.rateBytesPerSec = rateBytesPerSec
        self.etaSeconds = etaSeconds
    }
}

public struct ParsedProgress: Equatable, Sendable {
    public let percent: Double          // 0...1
    public let fileName: String?
    public let rateBytesPerSec: Int64?
    public init(percent: Double, fileName: String?, rateBytesPerSec: Int64?) {
        self.percent = percent
        self.fileName = fileName
        self.rateBytesPerSec = rateBytesPerSec
    }
}

public struct TransferAggregator {
    public let totalBytes: Int64
    public let totalFiles: Int

    private var completedBytes: Int64 = 0
    private var currentFileName: String?
    private var currentFileIndex: Int = 0
    private var currentFileSize: Int64 = 0
    private var currentFileBytesSeen: Int64 = 0
    private var currentFileMaxPct: Double = 0
    private var fileSizeLookup: [String: Int64] = [:]
    private var phase: TransferPhase = .preparing
    private var rateBytesPerSec: Int64?
    private var etaSeconds: Int?

    public init(totalBytes: Int64, totalFiles: Int, fileSizeLookup: [String: Int64] = [:]) {
        self.totalBytes = totalBytes
        self.totalFiles = totalFiles
        self.fileSizeLookup = fileSizeLookup
    }

    public var progress: TransferProgress {
        let totalDone = min(completedBytes + currentFileBytesSeen, totalBytes)
        return TransferProgress(
            phase: phase,
            totalBytes: totalBytes,
            completedBytes: totalDone,
            totalFiles: totalFiles,
            currentFileIndex: currentFileIndex,
            currentFileName: currentFileName,
            rateBytesPerSec: rateBytesPerSec,
            etaSeconds: etaSeconds
        )
    }

    public mutating func startSending() {
        if case .preparing = phase { phase = .sending }
    }

    public mutating func ingest(_ event: ParsedProgress) {
        // Only meaningful once sending.
        if case .preparing = phase { phase = .sending }

        if let name = event.fileName, name != currentFileName {
            // Close out previous file (use the max-pct snapshot we kept).
            completedBytes += currentFileSize  // full size of the previous file
            currentFileIndex = min(currentFileIndex + 1, max(totalFiles, 1))
            currentFileName = name
            currentFileSize = fileSizeLookup[name] ?? 0
            currentFileBytesSeen = 0
            currentFileMaxPct = 0
        } else if currentFileName == nil {
            // First event with no name; just remember name if we have it.
            currentFileName = event.fileName
            if let n = event.fileName { currentFileSize = fileSizeLookup[n] ?? 0 }
        }

        let clamped = min(max(event.percent, 0), 1)
        if clamped >= currentFileMaxPct {
            currentFileMaxPct = clamped
            currentFileBytesSeen = Int64(Double(currentFileSize) * clamped)
        }

        rateBytesPerSec = event.rateBytesPerSec
        if let r = event.rateBytesPerSec, r > 0 {
            let remaining = max(totalBytes - (completedBytes + currentFileBytesSeen), 0)
            etaSeconds = Int(remaining / r)
        }
    }

    public mutating func complete() {
        completedBytes = totalBytes
        currentFileBytesSeen = 0
        currentFileMaxPct = 1
        phase = .done
    }

    public mutating func fail(_ message: String) {
        phase = .failed(message)
    }
}
```

```swift
import Foundation
import Combine
import FastSCPCore

@MainActor
final class TransferTracker: ObservableObject {
    @Published private(set) var progress: TransferProgress

    private let selections: [URL]
    private var aggregator: TransferAggregator
    private var preScanDone = false

    init(selections: [URL]) {
        self.selections = selections
        // Provisional initial values; prepare() updates once scan completes.
        self.aggregator = TransferAggregator(totalBytes: 0, totalFiles: 0, fileSizeLookup: [:])
        self.progress = aggregator.progress
    }

    /// Background recursive scan to compute totalBytes + per-file lookup.
    /// Safe to call multiple times (no-op after first).
    func prepare() async {
        guard !preScanDone else { return }
        preScanDone = true
        let sels = selections
        let result = await Task.detached(priority: .userInitiated) {
            LocalScanner.scan(urls: sels)
        }.value
        aggregator = TransferAggregator(
            totalBytes: result.totalBytes,
            totalFiles: result.fileCount,
            fileSizeLookup: result.lookup
        )
        progress = aggregator.progress
    }

    func start() {
        aggregator.startSending()
        progress = aggregator.progress
    }

    /// Called from the SSHExecutor progress callback (off main; we hop to main here).
    func ingest(_ event: SCPProgress?) {
        guard let p = event else { return }       // nil = end-of-stream signal; we use try/catch for definitive outcome
        let parsed = ParsedProgress(
            percent: Double(p.percent) / 100.0,
            fileName: p.fileName,
            rateBytesPerSec: p.rateBytesPerSec
        )
        aggregator.ingest(parsed)
        progress = aggregator.progress
    }

    func complete() {
        aggregator.complete()
        progress = aggregator.progress
    }

    func fail(_ message: String) {
        aggregator.fail(message)
        progress = aggregator.progress
    }
}

private struct LocalScanResult {
    let totalBytes: Int64
    let fileCount: Int
    let lookup: [String: Int64]   // keyed by basename; collisions last-wins is acceptable for progress UI
}

private enum LocalScanner {
    static func scan(urls: [URL]) -> LocalScanResult {
        var total: Int64 = 0
        var count = 0
        var lookup: [String: Int64] = [:]
        let fm = FileManager.default
        let keys: [URLResourceKey] = [.isRegularFileKey, .isDirectoryKey, .isSymbolicLinkKey, .fileSizeKey]
        for root in urls {
            count += 1
            if let e = fm.enumerator(at: root, includingPropertiesForKeys: keys, options: [.skipsHiddenFiles]) {
                for case let url as URL in e {
                    let rv = try? url.resourceValues(forKeys: Set(keys))
                    let isSymlink = rv?.isSymbolicLink ?? false
                    let isDir = rv?.isDirectory ?? false
                    if isSymlink || isDir { continue }   // dirs not counted; symlinks size = link itself, ignored for simplicity
                    let size = Int64(rv?.fileSize ?? 0)
                    total += size
                    lookup[url.lastPathComponent] = size
                }
            }
        }
        return LocalScanResult(totalBytes: total, fileCount: count, lookup: lookup)
    }
}
```

```swift
import AppKit
import SwiftUI
import FastSCPCore

@MainActor
final class QuickTransferHUDController {
    private var window: NSPanel?
    private var closeTask: Task<Void, Never>?
    private let tracker: TransferTracker
    private let alias: String
    private let path: String

    init(tracker: TransferTracker, alias: String, path: String) {
        self.tracker = tracker
        self.alias = alias
        self.path = path
    }

    func show() {
        let host = NSHostingController(rootView: HUDView(tracker: tracker, alias: alias, path: path) { [weak self] in
            self?.dismiss()
        })
        host.view.translatesAutoresizingMaskIntoConstraints = false

        let panel = NSPanel(contentViewController: host)
        panel.styleMask = [.borderless, .resizable]
        panel.isFloatingPanel = true
        panel.level = .floating
        panel.hidesOnDeactivate = false
        panel.becomesKeyOnlyIfNeeded = true
        panel.collectionBehavior = [.canJoinAllSpaces, .fullScreenAuxiliary]
        panel.isReleasedWhenClosed = false
        panel.titlebarAppearsTransparent = true
        panel.titleVisibility = .hidden
        panel.backgroundColor = .clear
        panel.hasShadow = true
        panel.setContentSize(NSSize(width: 320, height: 110))
        positionTopRight(panel)

        self.window = panel
        panel.orderFrontRegardless()
        NSApp.activate(ignoringOtherApps: false)

        observeCompletion()
    }

    private func positionTopRight(_ panel: NSPanel) {
        guard let screen = NSScreen.main else { return }
        let v = screen.visibleFrame
        let size = panel.frame.size
        let x = v.maxX - size.width - 16
        let y = v.maxY - size.height - 16
        panel.setFrameOrigin(NSPoint(x: x, y: y))
    }

    private func observeCompletion() {
        // Poll with a Combine subscription on @Published progress.
        // We use a simple Task loop here to avoid Combine import in this file.
        Task { @MainActor [weak self] in
            while let self, let panel = self.window {
                switch self.tracker.progress.phase {
                case .done:
                    panel.makeKeyAndOrderFront(nil)
                    try? await Task.sleep(nanoseconds: 1_200_000_000)
                    self.fadeOutAndDismiss()
                    return
                case .failed:
                    // Stay until user closes (button in view).
                    return
                default:
                    try? await Task.sleep(nanoseconds: 250_000_000)
                    _ = panel // silence unused
                }
            }
        }
    }

    private func fadeOutAndDismiss() {
        guard let panel = window else { return }
        NSAnimationContext.runAnimationGroup { ctx in
            ctx.duration = 0.25
            panel.animator().alphaValue = 0
        } completionHandler: { [weak self] in
            self?.dismiss()
        }
    }

    func dismiss() {
        closeTask?.cancel()
        window?.orderOut(nil)
        window = nil
    }
}

private struct HUDView: View {
    @ObservedObject var tracker: TransferTracker
    let alias: String
    let path: String
    var onClose: () -> Void

    var body: some View {
        VStack(alignment: .leading, spacing: 8) {
            switch tracker.progress.phase {
            case .preparing:
                progressBody(title: "准备中 \(alias):\(path)",
                             percent: 0,
                             sizeText: "正在统计文件…",
                             rateText: nil,
                             etaText: nil,
                             fileText: nil)
            case .sending:
                progressBody(title: "正在传输 \(alias):\(path)",
                             percent: tracker.progress.percent,
                             sizeText: "\(ByteFormat.size(tracker.progress.completedBytes)) / \(ByteFormat.size(tracker.progress.totalBytes))",
                             rateText: tracker.progress.rateBytesPerSec.map(ByteFormat.rate),
                             etaText: etaText,
                             fileText: fileText)
            case .done:
                doneBody
            case .failed(let msg):
                failedBody(msg: msg)
            }
        }
        .padding(12)
        .frame(width: 320)
        .background(
            RoundedRectangle(cornerRadius: 10, style: .continuous)
                .fill(.regularMaterial)
        )
        .overlay(
            RoundedRectangle(cornerRadius: 10, style: .continuous)
                .stroke(Color.black.opacity(0.1), lineWidth: 0.5)
        )
    }

    @ViewBuilder
    private func progressBody(title: String, percent: Double, sizeText: String, rateText: String?, etaText: String?, fileText: String?) -> some View {
        HStack(spacing: 6) {
            Image(systemName: "arrow.up.circle").foregroundStyle(.tint)
            Text(title).font(.system(size: 12, weight: .medium)).lineLimit(1).truncationMode(.middle)
            Spacer()
            Text("\(Int(percent * 100))%").font(.caption).monospacedDigit()
        }
        ProgressView(value: percent).progressViewStyle(.linear)
        HStack(spacing: 6) {
            Text(sizeText).font(.caption2).monospacedDigit()
            if let rate = rateText { Text("· \(rate)").font(.caption2).monospacedDigit() }
            Spacer()
            if let eta = etaText { Text(eta).font(.caption2).monospacedDigit().foregroundStyle(.secondary) }
        }
        if let fileText {
            Text(fileText).font(.caption2).foregroundStyle(.secondary).lineLimit(1).truncationMode(.middle)
        }
    }

    @ViewBuilder
    private var doneBody: some View {
        HStack(spacing: 8) {
            Image(systemName: "checkmark.circle.fill").foregroundStyle(.green)
            Text("已发送到 \(alias):\(path)").font(.system(size: 12, weight: .medium)).lineLimit(1).truncationMode(.middle)

    @ViewBuilder
    private func failedBody(msg: String) -> some View {

                Text("传输失败").font(.system(size: 12, weight: .semibold))

            Text(msg).font(.caption2).foregroundStyle(.secondary).lineLimit(3)
            HStack {
                Spacer()
                Button("关闭") { onClose() }.controlSize(.small)
            }
        }
    }

    private var etaText: String? {
        guard let s = tracker.progress.etaSeconds, s > 0 else { return nil }
        if s >= 3600 { return "剩余 > 1 小时" }
        let m = s / 60, r = s % 60
        return m > 0 ? "剩余 \(m) 分 \(r) 秒" : "剩余 \(r) 秒"

    private var fileText: String? {
        let i = tracker.progress.currentFileIndex
        let n = tracker.progress.totalFiles
        guard n > 0, i > 0 else { return nil }
        return "第 \(i) / \(n) 个 · \(tracker.progress.currentFileName ?? "")"
    }
```
